const CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTQPFnmAZG7QiOJTpcCWUSZQbkN90EAJfJQaf8BacjBBRkNzloun10HnMdLBzFWZt-qU4JHZaVb3I80/pub?gid=1567052873&single=true&output=csv"; async function loadLeaderboard(){ const response = await fetch(CSV_URL); const csv = await response.text(); const rows = csv.trim().split("\n"); const headers = rows[0].split(","); console.log(headers); const nameIndex = headers.indexOf("Name"); const totalIndex = headers.indexOf("Total"); const projectionIndex = headers.indexOf("Golden Boot Projection"); const goalsIndex = headers.indexOf("Projected Goals"); const data = rows.slice(1) .map(row => { const cols = row.split(","); return { name: cols[nameIndex]?.trim(), total: Number(cols[totalIndex]) || 0, projection: cols[projectionIndex]?.trim(), goals: cols[goalsIndex]?.trim() }; }) .filter(entry => entry.name); data.sort((a,b)=>b.total-a.total); const tbody = document.getElementById("leaderboard"); tbody.innerHTML = ""; data.forEach((entry,index)=>{ let rankClass = ""; if(index === 0) rankClass = "rank1"; if(index === 1) rankClass = "rank2"; if(index === 2) rankClass = "rank3"; tbody.innerHTML += ` ${index+1} ${entry.name} ${entry.total} ${entry.projection} ${entry.goals} `; }); document.getElementById("updated").innerText = "Last Updated: " + new Date().toLocaleString(); } loadLeaderboard(); setInterval(loadLeaderboard,300000);
