const CSV_URL = "https://docs.google.com/spreadsheets/d/e/2PACX-1vTQPFnmAZG7QiOJTpcCWUSZQbkN90EAJfJQaf8BacjBBRkNzloun10HnMdLBzFWZt-qU4JHZaVb3I80/pub?gid=1567052873&single=true&output=csv"; async function loadLeaderboard() { try { const response = await fetch(CSV_URL); const csv = await response.text(); const rows = csv.trim().split("\n").map(r => r.split(",")); const headers = rows[0].map(h => h.trim()); const nameIndex = headers.indexOf("Name"); const totalIndex = headers.indexOf("Total"); const projectionIndex = headers.indexOf("Golden Boot Projection"); const goalsIndex = headers.findIndex(h => h.toLowerCase().includes("projected")); const data = rows.slice(1) .map(row => ({ name: row[nameIndex]?.trim(), total: Number(row[totalIndex]) || 0, projection: row[projectionIndex]?.trim() || "", goals: row[goalsIndex]?.trim() || "" })) .filter(r => r.name && r.name !== "Name"); data.sort((a, b) => b.total - a.total); const tbody = document.getElementById("leaderboard"); tbody.innerHTML = ""; data.forEach((entry, i) => { let rankClass = ""; if (i === 0) rankClass = "rank1"; else if (i === 1) rankClass = "rank2"; else if (i === 2) rankClass = "rank3"; tbody.innerHTML += ` ${i + 1} ${entry.name} ${entry.total} ${entry.projection} ${entry.goals} `; }); document.getElementById("updated").innerText = "Last Updated: " + new Date().toLocaleString(); } catch (err) { console.error(err); document.getElementById("updated").innerText = "Error loading leaderboard (check console)"; } } loadLeaderboard(); setInterval(loadLeaderboard, 300000);
